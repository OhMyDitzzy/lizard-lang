function(lz_add_test name)
    add_executable(${name} ${name}.c)
    target_link_libraries(${name} PRIVATE lizard_frontend lizard_builtin lizard_core)
    lz_enable_warnings(${name})
    add_test(NAME ${name} COMMAND ${name})
endfunction()
